# Синтаксис TypeScript для Kotlin/Swift разработчика

#typescript #syntax #kotlin #swift

## Типы и переменные

```typescript
// TypeScript              // Kotlin                  // Swift
let name: string = "hi";  // var name: String = "hi"  // var name: String = "hi"
const count: number = 42;  // val count: Int = 42      // let count: Int = 42
let active: boolean = true; // var active: Boolean      // var active: Bool

// ПРАВИЛО:
// let   = var (мутабельная)
// const = val (Kotlin) / let (Swift) (иммутабельная)
```

### Вывод типов (type inference)

```typescript
const x = 42;        // number (выводится автоматически)
const s = "hello";   // string
// Работает как в Kotlin/Swift — тип определяется из значения
```

### Основные типы

| TypeScript | Kotlin | Swift |
|---|---|---|
| `string` | `String` | `String` |
| `number` | `Int` / `Double` | `Int` / `Double` |
| `boolean` | `Boolean` | `Bool` |
| `any` | `Any` | `Any` |
| `unknown` | `Any?` с проверкой | — |
| `void` | `Unit` | `Void` |
| `null` | `null` | `nil` |
| `undefined` | — (нет аналога) | — |
| `never` | `Nothing` | `Never` |

> [!warning] null vs undefined
> TypeScript имеет **два** "пустых" значения:
> - `null` — явно "ничего" (как null/nil)
> - `undefined` — "не задано" (переменная объявлена но не инициализирована)

## Nullable (Optionals)

```typescript
// TypeScript                          // Kotlin               // Swift
let value: string | null = null;       // var value: String?    // var value: String?
let maybe: string | undefined;         // (нет аналога)        // (нет аналога)
let either: string | null | undefined; // var value: String?    // var value: String?

// Optional chaining (одинаково!)
value?.length                          // value?.length         // value?.count

// Nullish coalescing
const result = value ?? "default";     // value ?: "default"   // value ?? "default"

// Non-null assertion (осторожно!)
const len = value!.length;             // value!!.length       // value!.count
```

## Функции

```typescript
// Обычная функция
function add(a: number, b: number): number {
  return a + b;
}
// Kotlin: fun add(a: Int, b: Int): Int = a + b
// Swift:  func add(_ a: Int, _ b: Int) -> Int { a + b }

// Стрелочная функция (лямбда) — САМЫЙ ЧАСТЫЙ ПАТТЕРН в этом проекте
const add = (a: number, b: number): number => a + b;
// Kotlin: val add = { a: Int, b: Int -> a + b }
// Swift:  let add = { (a: Int, b: Int) -> Int in a + b }

// Стрелочная функция с телом
const greet = (name: string): void => {
  console.log(`Hello ${name}`);
};

// Параметры по умолчанию
function create(name: string, active: boolean = true): void { }
// Идентично Kotlin/Swift

// Rest parameters (variadic)
function sum(...numbers: number[]): number { return numbers.reduce((a, b) => a + b); }
// Kotlin: fun sum(vararg numbers: Int): Int
// Swift:  func sum(_ numbers: Int...) -> Int
```

## Async / Await

```typescript
// TypeScript                                 // Kotlin (coroutines)
async function fetchData(): Promise<string> { // suspend fun fetchData(): String
  const result = await someApiCall();          //   val result = someApiCall()
  return result;                               //   return result
}

// Вызов
const data = await fetchData();                // val data = fetchData()

// Promise = Deferred (Kotlin) / Future (Combine/Swift)
// await = .await() (Kotlin) / await (Swift 5.5)
```

## Классы и интерфейсы

```typescript
// Interface (= Kotlin interface / Swift protocol)
export default interface IIntegration {
  provide(...args: unknown[]): void;
  enable(): void;
  disable(): void;
  getYTMScripts(): { name: string; script: string }[];
}

// Class с generics
export default class MemoryStore<T extends Record<string, unknown>> {
  private state: Record<string, unknown>;  // private var state: Map<String, Any?>
  private eventEmitter = new EventEmitter();

  constructor() {                           // init { } (Kotlin/Swift)
    this.state = {};
  }

  public get(key: string): unknown {
    return this.state[key];
  }

  public set(key: string, value: unknown) {
    const oldState = structuredClone(this.state);
    this.state[key] = value;
    this.eventEmitter.emit("stateChanged", this.state, oldState);
  }
}
```

| `private` | `private` | `private` |

## Импорты и экспорты

```typescript
// Named import/export
import { app, BrowserWindow } from "electron";
export function helper() { }
export { MyClass, helper };

// Default import/export
import log from "electron-log";
export default class MyClass { }

// Импорт типов
import type { StoreSchema } from "~shared/store/schema";

// Path aliases (настроены в tsconfig.json)
import { StoreSchema } from "~shared/store/schema";  // → src/shared/store/schema
import "~assets/app.css";                             // → src/assets/app.css
```

> [!note] Аналогия
> `import { X } from "module"` = `import module.X` (Kotlin) / `import Module` (Swift)
> `export` = `public` (делает доступным для импорта из других файлов)
